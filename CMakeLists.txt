cmake_minimum_required(VERSION 3.16)
project(JaliVulkan VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
add_library(jali_pipeline src/pipeline.cpp src/phonemes.cpp)
target_include_directories(jali_pipeline PUBLIC include)
target_compile_features(jali_pipeline PUBLIC cxx_std_17)
option(JALI_ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(JALI_ENABLE_SANITIZERS)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" AND NOT MSVC)
    target_compile_options(jali_pipeline PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(jali_pipeline PUBLIC -fsanitize=address,undefined)
  else()
    message(FATAL_ERROR "JALI_ENABLE_SANITIZERS requires GCC or Clang with sanitizer support")
  endif()
endif()
if(MSVC)
  target_compile_options(jali_pipeline PRIVATE /W4)
else()
  target_compile_options(jali_pipeline PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_library(jali_backends INTERFACE)
target_link_libraries(jali_backends INTERFACE jali_pipeline)
option(JALI_ENABLE_ESPEAK "Build the real eSpeak NG speech provider" OFF)
option(JALI_ENABLE_VULKAN "Build the native offscreen Vulkan face renderer" OFF)
if(JALI_ENABLE_ESPEAK)
  find_path(ESPEAK_INCLUDE_DIR espeak-ng/speak_lib.h)
  find_library(ESPEAK_LIBRARY NAMES espeak-ng)
  if(NOT ESPEAK_INCLUDE_DIR OR NOT ESPEAK_LIBRARY)
    message(FATAL_ERROR "Install eSpeak NG development files or set ESPEAK_INCLUDE_DIR and ESPEAK_LIBRARY")
  endif()
  find_package(Threads REQUIRED)
  add_library(jali_espeak src/espeak_voice.cpp)
  target_include_directories(jali_espeak PRIVATE ${ESPEAK_INCLUDE_DIR})
  target_link_libraries(jali_espeak PUBLIC jali_pipeline PRIVATE ${ESPEAK_LIBRARY} Threads::Threads)
  target_link_libraries(jali_backends INTERFACE jali_espeak)
  target_compile_definitions(jali_backends INTERFACE JALI_HAS_ESPEAK=1)
endif()
if(JALI_ENABLE_VULKAN)
  find_package(Vulkan REQUIRED)
  set(FACE_SHADER "${CMAKE_CURRENT_BINARY_DIR}/shaders/face.comp.spv")
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/shaders")
  find_program(GLSLANG_VALIDATOR glslangValidator)
  if(GLSLANG_VALIDATOR)
    add_custom_command(OUTPUT "${FACE_SHADER}"
      COMMAND ${GLSLANG_VALIDATOR} -V "${CMAKE_CURRENT_SOURCE_DIR}/shaders/face.comp" -o "${FACE_SHADER}"
      DEPENDS shaders/face.comp VERBATIM)
  else()
    find_program(NODE_EXECUTABLE node)
    if(NOT NODE_EXECUTABLE OR NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tools/node_modules/@webgpu/glslang")
      message(FATAL_ERROR "Install glslangValidator or run npm ci --prefix tools for the WASM shader compiler")
    endif()
    add_custom_command(OUTPUT "${FACE_SHADER}"
      COMMAND ${NODE_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tools/compile_shader.cjs"
        "${CMAKE_CURRENT_SOURCE_DIR}/shaders/face.comp" "${FACE_SHADER}"
      DEPENDS shaders/face.comp tools/compile_shader.cjs VERBATIM)
  endif()
  set(MESH_SHADERS "${FACE_SHADER}")
  foreach(SOURCE_SHADER skin.comp mesh.vert mesh.frag)
    set(OUTPUT_SHADER "${CMAKE_CURRENT_BINARY_DIR}/shaders/${SOURCE_SHADER}.spv")
    if(GLSLANG_VALIDATOR)
      add_custom_command(OUTPUT "${OUTPUT_SHADER}"
        COMMAND ${GLSLANG_VALIDATOR} -V "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${SOURCE_SHADER}" -o "${OUTPUT_SHADER}"
        DEPENDS "shaders/${SOURCE_SHADER}" VERBATIM)
    else()
      add_custom_command(OUTPUT "${OUTPUT_SHADER}"
        COMMAND ${NODE_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tools/compile_shader.cjs"
          "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${SOURCE_SHADER}" "${OUTPUT_SHADER}"
        DEPENDS "shaders/${SOURCE_SHADER}" tools/compile_shader.cjs VERBATIM)
    endif()
    list(APPEND MESH_SHADERS "${OUTPUT_SHADER}")
  endforeach()
  add_custom_target(jali_face_shader DEPENDS ${MESH_SHADERS})
  find_path(TINYGLTF_INCLUDE_DIR tiny_gltf.h HINTS "${CMAKE_CURRENT_SOURCE_DIR}/.cache/deps/tinygltf-2.9.3")
  if(NOT TINYGLTF_INCLUDE_DIR)
    message(FATAL_ERROR "Set TINYGLTF_INCLUDE_DIR to a tinygltf 2.9.3 checkout (bootstrap helper downloads it)")
  endif()
  find_package(Python3 3.8 REQUIRED COMPONENTS Interpreter)
  set(JALI_CHARACTER "${CMAKE_CURRENT_BINARY_DIR}/assets/android.glb")
  add_custom_command(OUTPUT "${JALI_CHARACTER}"
    COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tools/make_character.py" "${JALI_CHARACTER}"
    DEPENDS tools/make_character.py VERBATIM)
  set(JALI_LARGE_CHARACTER "${CMAKE_CURRENT_BINARY_DIR}/assets/android-67.glb")
  add_custom_command(OUTPUT "${JALI_LARGE_CHARACTER}"
    COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tools/make_character.py" "${JALI_LARGE_CHARACTER}" --joints 67
    DEPENDS tools/make_character.py VERBATIM)
  set(JALI_MAX_CHARACTER "${CMAKE_CURRENT_BINARY_DIR}/assets/android-128.glb")
  add_custom_command(OUTPUT "${JALI_MAX_CHARACTER}"
    COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tools/make_character.py" "${JALI_MAX_CHARACTER}" --joints 128
    DEPENDS tools/make_character.py VERBATIM)
  set(JALI_RICH_FIXTURE "${CMAKE_CURRENT_BINARY_DIR}/assets/rich-fixture.glb")
  add_custom_command(OUTPUT "${JALI_RICH_FIXTURE}"
    COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tools/make_rich_fixture.py" "${JALI_RICH_FIXTURE}"
    DEPENDS tools/make_rich_fixture.py VERBATIM)
  add_custom_target(jali_character_asset DEPENDS "${JALI_RICH_FIXTURE}" "${JALI_CHARACTER}" "${JALI_LARGE_CHARACTER}" "${JALI_MAX_CHARACTER}")
  add_library(jali_vulkan src/vulkan_renderer.cpp src/character.cpp)
  target_include_directories(jali_vulkan PRIVATE ${TINYGLTF_INCLUDE_DIR})
  add_dependencies(jali_vulkan jali_face_shader jali_character_asset)
  target_link_libraries(jali_vulkan PUBLIC jali_pipeline PRIVATE Vulkan::Vulkan)
  add_executable(jali_rich src/rich.cpp)
  target_include_directories(jali_rich PRIVATE ${TINYGLTF_INCLUDE_DIR})
  target_link_libraries(jali_rich PRIVATE jali_backends)
  target_link_libraries(jali_backends INTERFACE jali_vulkan)
  target_compile_definitions(jali_backends INTERFACE JALI_HAS_VULKAN=1 JALI_DEFAULT_FACE_SHADER="${FACE_SHADER}" JALI_DEFAULT_CHARACTER="${JALI_CHARACTER}" JALI_LARGE_CHARACTER="${JALI_LARGE_CHARACTER}" JALI_MAX_CHARACTER="${JALI_MAX_CHARACTER}" JALI_RICH_FIXTURE="${JALI_RICH_FIXTURE}")
endif()
if(JALI_ENABLE_ESPEAK AND JALI_ENABLE_VULKAN)
  add_executable(jali_live src/live.cpp)
  target_include_directories(jali_live PRIVATE ${TINYGLTF_INCLUDE_DIR})
  target_link_libraries(jali_live PRIVATE jali_backends)
endif()
add_executable(jali_demo src/main.cpp)
target_link_libraries(jali_demo PRIVATE jali_backends)
if(BUILD_TESTING)
  add_executable(jali_phoneme_tests tests/phoneme_tests.cpp)
  target_link_libraries(jali_phoneme_tests PRIVATE jali_pipeline)
  add_test(NAME phoneme_contracts COMMAND jali_phoneme_tests)
  add_executable(jali_tests tests/pipeline_tests.cpp)
  target_link_libraries(jali_tests PRIVATE jali_pipeline)
  add_test(NAME pipeline_contracts COMMAND jali_tests)
  add_test(NAME demo_fixture COMMAND jali_demo --output ${CMAKE_CURRENT_BINARY_DIR}/fixture)
  set_tests_properties(demo_fixture PROPERTIES FIXTURES_SETUP audio_fixture)
  if(JALI_ENABLE_VULKAN)
    add_executable(jali_character_tests tests/character_tests.cpp)
    target_include_directories(jali_character_tests PRIVATE ${TINYGLTF_INCLUDE_DIR})
    target_link_libraries(jali_character_tests PRIVATE jali_backends)
    add_test(NAME malformed_character_contracts COMMAND jali_character_tests)
    find_program(NODE_EXECUTABLE node)
    if(NODE_EXECUTABLE AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tools/node_modules/gltf-validator")
      add_test(NAME gltf_spec_validation COMMAND ${NODE_EXECUTABLE}
        "${CMAKE_CURRENT_SOURCE_DIR}/tools/validate_character.cjs" "${JALI_CHARACTER}")
      add_test(NAME gltf_rich_fixture_validation COMMAND ${NODE_EXECUTABLE}
        "${CMAKE_CURRENT_SOURCE_DIR}/tools/validate_character.cjs" "${JALI_RICH_FIXTURE}")
      add_test(NAME gltf_67_joint_validation COMMAND ${NODE_EXECUTABLE}
        "${CMAKE_CURRENT_SOURCE_DIR}/tools/validate_character.cjs" "${JALI_LARGE_CHARACTER}")
      add_test(NAME gltf_128_joint_validation COMMAND ${NODE_EXECUTABLE}
        "${CMAKE_CURRENT_SOURCE_DIR}/tools/validate_character.cjs" "${JALI_MAX_CHARACTER}")
    endif()
  endif()
  if(JALI_ENABLE_ESPEAK OR JALI_ENABLE_VULKAN)
    add_executable(jali_backend_tests tests/backend_tests.cpp)
    target_link_libraries(jali_backend_tests PRIVATE jali_backends)
    add_test(NAME real_backend_contracts COMMAND jali_backend_tests)
  endif()
  find_package(Python3 3.8 COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    if(TARGET jali_live)
      add_test(NAME native_live_protocol COMMAND ${Python3_EXECUTABLE}
        "${CMAKE_CURRENT_SOURCE_DIR}/tools/smoke_native.py" --binary $<TARGET_FILE:jali_live>
        --output "${CMAKE_CURRENT_BINARY_DIR}/live-protocol")
    endif()
    add_test(NAME serialized_sync COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/analyze_sync.py
      --audio ${CMAKE_CURRENT_BINARY_DIR}/fixture/session.wav
      --curves ${CMAKE_CURRENT_BINARY_DIR}/fixture/face.csv --max-lag-ms 100)
    set_tests_properties(serialized_sync PROPERTIES FIXTURES_REQUIRED audio_fixture)
    add_test(NAME python_contracts COMMAND ${Python3_EXECUTABLE} -m unittest discover
      -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -p "test_*.py")
  endif()
endif()
